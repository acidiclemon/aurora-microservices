# 1. Get Token from Okta (Replace placeholders with your data)
curl --request POST \
  --url https://your-domain.okta.com/oauth2/default/v1/token \
  --header 'accept: application/json' \
  --header 'content-type: application/x-www-form-urlencoded' \
  --data 'grant_type=client_credentials' \
  --data 'client_id=YOUR_CLIENT_ID' \
  --data 'client_secret=YOUR_CLIENT_SECRET' \
  --data 'scope=test'


# 2. Hit the Envoy-protected API using the token string from above
curl -i https://api.your-domain.com/ \
  -H "Authorization: Bearer YOUR_ACCESS_TOKEN_STRING"
